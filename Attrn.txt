  <script>
    let currentAppType = 'Adoption';

    function setAppType(type) {
      currentAppType = type;
      const toggleAdopt = document.getElementById('toggle-adopt');
      const toggleFoster = document.getElementById('toggle-foster');
      const submitBtn = document.getElementById('submit-btn');
      const petGroup = document.getElementById('pet-select-group');

      if (type === 'Adoption') {
        toggleAdopt.classList.add('active');
        toggleFoster.classList.remove('active');
        submitBtn.innerText = 'Submit Adoption Application';
        petGroup.style.display = 'flex';
      } else {
        toggleFoster.classList.add('active');
        toggleAdopt.classList.remove('active');
        submitBtn.innerText = 'Submit Foster Application';
        // Foster applications are often general
        document.getElementById('target-pet').value = 'General Interest';
      }
    }

    function selectPet(petName) {
      setAppType('Adoption');
      document.getElementById('target-pet').value = petName;
      document.getElementById('apply').scrollIntoView({ behavior: 'smooth' });
    }

    function handleFormSubmit(e) {
      e.preventDefault();
      const name = document.getElementById('first-name').value;
      const modalText = document.getElementById('modal-text');
      
      modalText.innerText = `Thank you, ${name}! Your ${currentAppType.toLowerCase()} application has been submitted successfully. Our coordinators will review your details and reach out within 24–48 hours.`;
      
      document.getElementById('success-modal').style.display = 'flex';
      document.getElementById('rescue-application').reset();
      setAppType('Adoption');
    }

    function closeModal() {
      document.getElementById('success-modal').style.display = 'none';
    }
  </script>
